cmake_minimum_required(VERSION 3.16)
project(circuit_lab VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(ENGINE_SOURCES src/engine/src/engine.cpp src/engine/src/catalog.cpp src/engine/src/bridge.cpp)
if(EMSCRIPTEN)
  add_executable(signal_engine ${ENGINE_SOURCES})
  set_target_properties(signal_engine PROPERTIES OUTPUT_NAME engine SUFFIX .mjs)
  target_compile_options(signal_engine PRIVATE -fexceptions)
  target_link_options(signal_engine PRIVATE
    --no-entry
    -sMODULARIZE=1 -sEXPORT_ES6=1 -sEXPORT_NAME=createCircuitEngine
    -sENVIRONMENT=web,worker,node -sFILESYSTEM=0
    -sALLOW_MEMORY_GROWTH=1 -sMAXIMUM_MEMORY=268435456
    -sDISABLE_EXCEPTION_CATCHING=0
    "-sEXPORTED_RUNTIME_METHODS=['cwrap','setValue']"
    "-sEXPORTED_FUNCTIONS=['_circuit_catalog','_circuit_reference','_circuit_create','_circuit_destroy','_circuit_node','_circuit_wire','_circuit_state','_circuit_simulate','_circuit_check','_malloc','_free']")
else()
  add_library(signal_engine SHARED ${ENGINE_SOURCES})
endif()
target_include_directories(signal_engine PUBLIC src/engine/include)
if(MSVC)
  target_compile_options(signal_engine PRIVATE /W4)
else()
  target_compile_options(signal_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(SIGNAL_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(SIGNAL_SANITIZE)
  if(MSVC OR EMSCRIPTEN)
    message(FATAL_ERROR "SIGNAL_SANITIZE requires GCC or Clang")
  endif()
  target_compile_options(signal_engine PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(signal_engine PUBLIC -fsanitize=address,undefined)
endif()
if(NOT EMSCRIPTEN)
  add_executable(signal_demo examples/native.cpp)
  target_link_libraries(signal_demo PRIVATE signal_engine)
  include(CTest)
  if(BUILD_TESTING)
    add_executable(engine_tests tests/engine_tests.cpp)
    target_link_libraries(engine_tests PRIVATE signal_engine)
    add_test(NAME engine COMMAND engine_tests)
  endif()
endif()
