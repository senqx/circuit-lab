cmake_minimum_required(VERSION 3.16)
project(signal_garden VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(signal_engine SHARED engine/src/engine.cpp engine/src/catalog.cpp engine/src/bridge.cpp)
target_include_directories(signal_engine PUBLIC engine/include)
if(MSVC)
  target_compile_options(signal_engine PRIVATE /W4)
else()
  target_compile_options(signal_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(SIGNAL_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(SIGNAL_SANITIZE)
  if(MSVC)
    message(FATAL_ERROR "SIGNAL_SANITIZE requires GCC or Clang")
  endif()
  target_compile_options(signal_engine PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(signal_engine PUBLIC -fsanitize=address,undefined)
endif()
add_executable(signal_demo examples/native.cpp)
target_link_libraries(signal_demo PRIVATE signal_engine)
include(CTest)
if(BUILD_TESTING)
  add_executable(engine_tests tests/engine_tests.cpp)
  target_link_libraries(engine_tests PRIVATE signal_engine)
  add_test(NAME engine COMMAND engine_tests)
endif()
